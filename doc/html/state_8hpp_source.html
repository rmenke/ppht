157 bool next(point_t &point) {
158 auto begin = _pending.begin();
159 auto end = _pending.end();
160
161 end = std::remove_if(begin, end, [this](auto &&p) {
162 return status(p) != status_t::pending;
163 });
164
165 _pending.erase(end, _pending.end());
166
167 if (begin == end) return false;
168
169 using index_type = typename decltype(_pending)::difference_type;
170
171 std::uniform_int_distribution<index_type>
172 dist{0, std::distance(begin, end) - 1};
173
174 auto iter = begin + dist(_urbg);
175
176 point = std::exchange(*iter, *(--end));
177
178 auto &cell = _state[std::get<1>(point)][std::get<0>(point)];
179
180 assert(cell == status_t::pending);
181
182 cell = status_t::voted;
183
184 return true;

215template <template <class> class Raster>
216point_set scan(state<Raster> &s, segment_t const &segment, unsigned radius,
217 unsigned max_gap) {
218 // The initial gap is technically infinite, but anything
219 // larger than max_gap will do.
220 auto gap = max_gap + 1;
221
222 std::vector<point_set> point_sets;
223
224 auto const &p0 = std::get<0>(segment);
225 auto const &p1 = std::get<1>(segment);
226
227 auto const r = s.rows();
228 auto const c = s.cols();
229
230 for (auto const &[point, points] : channel(p0, p1, radius)) {
231 std::set<point_t> found;
232
233 for (auto const &p : points) {
234 if (p[0] < 0 || p[0] >= c) continue;
235 if (p[1] < 0 || p[1] >= r) continue;
236
237 auto status = s.status(p);
238
239 if (status == status_t::pending || status == status_t::voted) {
240 found.insert(p);
241 }
242 }
243
244 if (found.empty()) { // no hits
245 ++gap;
246 }
247 else {
248 // If the gap is too large to ignore, start a new point set.
249 if (gap > max_gap) point_sets.emplace_back();
250 point_sets.back().add_point(point, found);
251
252 gap = 0;
253 }
254 }
255
256 if (point_sets.empty()) {
257 throw std::logic_error{"channel contained no viable segments"};
258 }
259
260 auto longest = std::max_element(point_sets.begin(), point_sets.end());
261
262 return *longest;
263}
