158 bool next(point_t &point) {
159 auto begin = _pending.begin();
160 auto end = _pending.end();
161
162 end = std::remove_if(begin, end, [this](auto &&p) {
163 return status(p) != status_t::pending;
164 });
165
166 _pending.erase(end, _pending.end());
167
168 if (begin == end) return false;
169
170 using index_type = typename decltype(_pending)::difference_type;
171
172 std::uniform_int_distribution<index_type> dist{
173 0, std::distance(begin, end) - 1};
174
175 auto iter = begin + dist(_urbg);
176
177 point = std::exchange(*iter, *(--end));
178
179 auto &cell = _state[std::get<1>(point)][std::get<0>(point)];
180
181 assert(cell == status_t::pending);
182
183 cell = status_t::voted;
184
185 return true;
186 }

198 segment_t line_intersect(line_t const &line) const {
199 // There are a few degenerate cases where multiple matches for
200 // the same endpoint can be found, e.g., a line through the
201 // origin. Using a set eliminates most of these cases. See
202 // the comment at the end for what happens to those that slip
203 // through.
204
205 auto const &t = std::get<0>(line);
206 auto const &r = std::get<1>(line);
207
208 std::set<point_t> endpoints;
209
210 auto const &cost = std::get<0>(cossin[t]);
211 auto const &sint = std::get<1>(cossin[t]);
212
213 using limit = std::numeric_limits<long>;
214 static constexpr long lo = limit::min();
215 static constexpr long hi = limit::max();
216
217 auto get_x = [&](double y) -> long {
218 double x = std::rint((r - sint * y) / cost);
219 return x < lo ? lo : hi < x ? hi : x;
220 };
221
222 auto get_y = [&](double x) -> long {
223 double y = std::rint((r - cost * x) / sint);
224 return y < lo ? lo : hi < y ? hi : y;
225 };
226
227 long const w = cols() - 1;
228 long const h = rows() - 1;
229
230 long const x_min = get_x(0);
231 long const y_min = get_y(0);
232 long const x_max = get_x(h);
233 long const y_max = get_y(w);
234
235 if (0 <= y_min && y_min <= h) endpoints.emplace(0, y_min);
236 if (0 <= x_min && x_min <= w) endpoints.emplace(x_min, 0);
237 if (0 <= y_max && y_max <= h) endpoints.emplace(w, y_max);
238 if (0 <= x_max && x_max <= w) endpoints.emplace(x_max, h);
239
240 if (endpoints.empty()) {
241 throw std::logic_error{"line (" + std::to_string(t) + ", " +
242 std::to_string(r) +
243 ") does not intersect bitmap"};
244 }
245
246 // If endpoints.size() > 2, ignore the points in the middle.
247 // If endpoints.size() == 1, create single-pixel segment.
248
249 return segment_t{*endpoints.begin(), *endpoints.rbegin()};
250 }

280template <template <class> class Raster>
281point_set scan(state<Raster> &s, segment_t const &segment, unsigned radius,
282 unsigned max_gap) {
283 // The initial gap is technically infinite, but anything
284 // larger than max_gap will do.
285 auto gap = max_gap + 1;
286
287 std::vector<point_set> point_sets;
288
289 auto const &p0 = std::get<0>(segment);
290 auto const &p1 = std::get<1>(segment);
291
292 long const r = s.rows();
293 long const c = s.cols();
294
295 for (auto const &[point, points] : channel(p0, p1, radius)) {
296 std::set<point_t> found;
297
298 for (auto const &p : points) {
299 if (p[0] < 0 || p[0] >= c) continue;
300 if (p[1] < 0 || p[1] >= r) continue;
301
302 auto status = s.status(p);
303
304 if (status == status_t::pending || status == status_t::voted) {
305 found.insert(p);
306 }
307 }
308
309 if (found.empty()) { // no hits
310 ++gap;
311 }
312 else {
313 // If the gap is too large to ignore, start a new point set.
314 if (gap > max_gap) point_sets.emplace_back();
315 point_sets.back().add_point(point, found);
316
317 gap = 0;
318 }
319 }
320
321 if (point_sets.empty()) {
322 throw std::logic_error{"channel contained no viable segments"};
323 }
324
325 auto longest = std::max_element(point_sets.begin(), point_sets.end());
326
327 return *longest;
328}
