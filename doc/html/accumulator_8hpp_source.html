171 static std::pair<std::size_t, int>
172 rho_info(std::size_t rows, std::size_t cols) noexcept {
173 double diag = std::ceil(std::hypot(rows - 1, cols - 1));
174 int rho_exp = std::ilogb(max_theta / (diag * 2.0 + 1.0));
175
176 // lo is 2 * diag * (1 << rho_exp).
177
178 std::size_t lo = std::ceil(std::scalbn(diag, rho_exp + 1)) + 1;
179
180 // hi is 2 * diag * (2 << rho_exp). This is the smallest
181 // rho_exp such that 2 * diag * (1 << rho_exp) >= max_theta.
182
183 std::size_t hi = std::ceil(std::scalbn(diag, rho_exp + 2)) + 1;
184
185 assert(lo <= max_theta && max_theta <= hi);
186
187 // Select the rho_exp that makes the counter matrix the most
188 // square.
189
190 if ((max_theta - lo) <= (hi - max_theta)) {
191 return std::make_pair(lo, rho_exp);
192 }
193 else {
194 return std::make_pair(hi, rho_exp + 1);
195 }
196 }

264 template <class ForwardIt>
265 ForwardIt best_candidate(ForwardIt begin, ForwardIt end) {
266 auto best = std::gcd(std::get<0>(*begin), max_theta / 2);
267
268 for (auto iter = std::next(begin); iter != end; ++iter) {
269 auto gcd = std::gcd(std::get<0>(*iter), max_theta / 2);
270 if (best < gcd) {
271 begin = iter;
272 best = gcd;
273 }
274 }
275
276 return begin;
277 }

293 std::optional<line_t> vote(point_t const &p) {
294 auto const max_rho = _counters.rows();
295
296 Count n = _min_trigger_points;
297
298 std::vector<line_t> found;
299
300 std::size_t theta;
301 double rho;
302
303 // Increment the cells in the register, keeping track of the
304 // current maxima.
305
306 for (theta = 0; theta < max_theta; ++theta) {
307 rho = scale_rho(p.dot(cossin[theta]));
308 if (rho < 0 || rho >= max_rho) continue;
309
310 auto &counter = _counters[rho][theta];
311
312 ++counter;
313
314 // If we have found a new maxima, update n and discard the
315 // old candidates.
316
317 if (n < counter) {
318 n = counter;
319 found.clear();
320 }
321
322 if (n == counter) {
323 found.emplace_back(theta, unscale_rho(rho));
324 }
325 }
326
327 ++_votes;
328
329 // If we do not have a candidate line, stop.
330
331 if (found.empty()) return std::nullopt;
332
333 // There are max_theta * max_rho cells in the register.
334 // Each vote will increment max_theta of these cells, one
335 // per column.
336 //
337 // Assuming the null hypothesis (the image is random noise),
338 // E[n] = votes/max_rho for all cells in the register.
339
340 double const lambda = static_cast<double>(_votes) / max_rho;
341
342 // For the null hypothesis, the cells are filled (roughly)
343 // according to a Poisson model:
344 //
345 // p(n) = λⁿ/n!·exp(-λ)
346 // = λⁿ/Γ(n+1)·exp(-λ)
347 // ln p(n) = n·ln(λ) - lnΓ(n+1) - λ
348
349 double const lnp =
350 n * std::log(lambda) - std::lgamma(n + 1) - lambda;
351
352 // lnp is the log probability that a bin filled randomly would
353 // contain a count of n. If the probability is above the
354 // significance threshold, we assume that the bin was filled
355 // by noise and tell the caller we did not find a segment.
356
357 if (lnp >= _log_threshold) return std::nullopt;
358
359 // Reject the null hypothesis.
360
361 assert(!found.empty());
362
363 return *best_candidate(found.begin(), found.end());
364 }

371 void unvote(point_t const &p) {
372 auto const max_rho = _counters.rows();
373
374 for (std::size_t theta = 0; theta < max_theta; ++theta) {
375 double const rho = scale_rho(p.dot(cossin[theta]));
376 if (rho < 0 || rho >= max_rho) continue;
377
378 auto &counter = _counters[rho][theta];
379
380 if (counter == 0) {
381 throw std::logic_error{__func__};
382 }
383
384 --counter;
385 }
386
387 --_votes;
388 }
