87 template <std::size_t Major>
88 class axial_scanner : public scanner {
89 static constexpr std::size_t Minor = 1 - Major;
90
91 const point_t step;
92 const std::size_t radius;
93
94 public:
95 axial_scanner(point_t delta, std::size_t radius)
96 : step{signum(std::get<0>(delta)), signum(std::get<1>(delta))}
97 , radius(radius) {
98 assert(std::get<Major>(step) != 0);
99 assert(std::get<Minor>(step) == 0);
100 }
101
102 void fill(point_t point, std::set<point_t> &points) const override {
103 std::get<Minor>(point) -= radius;
104
105 for (std::size_t m = 1; m < 2 * radius; ++m) {
106 ++std::get<Minor>(point);
107 points.insert(point);
108 }
109 }
110
111 void advance(point_t &point) override {
112 std::get<Major>(point) += std::get<Major>(step);
113 }
114 };

133 template <std::size_t Major>
134 class bresenham_scanner : public scanner {
135 static constexpr std::size_t Minor = 1U - Major;
136
137 const point_t _delta, _step, _perp_step;
138
139 const double _width;
140
141 const long _threshold =
142 std::get<Major>(_delta) - 2 * std::get<Minor>(_delta);
143 const long _post_minor_move = -2 * std::get<Major>(_delta);
144 const long _post_major_move = 2 * std::get<Minor>(_delta);
145
146 long _error = 0, _phase = 0;
147
148 void perpendiculars(point_t point, std::set<point_t> &points,
149 long initial_phase,
150 long initial_error) const {
151 auto p = point;
152 auto phase = initial_phase;
153
154 const auto d =
155 std::get<Major>(_delta) + std::get<Minor>(_delta);
156
157 for (auto tk = d - initial_error; tk < _width;
158 tk -= _post_minor_move) {
159 points.insert(p);
160
161 if (phase >= _threshold) {
162 std::get<Major>(p) += std::get<Major>(_perp_step);
163 phase += _post_minor_move;
164 tk += _post_major_move;
165 }
166
167 std::get<Minor>(p) += std::get<Minor>(_perp_step);
168 phase += _post_major_move;
169 }
170
171 p = point;
172 phase = -initial_phase;
173
174 for (auto tk = d + initial_error; tk <= _width;
175 tk -= _post_minor_move) {
176 points.insert(p);
177
178 if (phase > _threshold) {
179 std::get<Major>(p) -= std::get<Major>(_perp_step);
180 phase += _post_minor_move;
181 tk += _post_major_move;
182 }
183
184 std::get<Minor>(p) -= std::get<Minor>(_perp_step);
185 phase += _post_major_move;
186 }
187 }
188
189 public:
190 bresenham_scanner(point_t delta, std::size_t radius)
191 : _delta{std::abs(std::get<0>(delta)),
192 std::abs(std::get<1>(delta))}
193 , _step{signum(std::get<0>(delta)), signum(std::get<1>(delta))}
194 , _perp_step{Major == 0 ? -std::get<0>(_step)
195 : +std::get<0>(_step),
196 Major == 0 ? +std::get<1>(_step)
197 : -std::get<1>(_step)}
198 , _width{2.0 * radius *
199 std::hypot(std::get<0>(_delta), std::get<1>(_delta))} {
200 }
201
202 void fill(point_t point, std::set<point_t> &points) const override {
203 perpendiculars(point, points, _phase, _error);
204
205 if (_error >= _threshold && _phase >= _threshold) {
206 std::get<Minor>(point) += std::get<Minor>(_step);
207 perpendiculars(
208 point, points,
209 (_phase + _post_minor_move + _post_major_move),
210 _error + _post_minor_move);
211 }
212
213 if (points.empty()) points.insert(point);
214 }
215
216 void advance(point_t &pt) override {
217 if (_error >= _threshold) {
218 std::get<Minor>(pt) += std::get<Minor>(_step);
219 _error += _post_minor_move;
220
221 if (_phase >= _threshold) { _phase += _post_minor_move; }
222
223 _phase += _post_major_move;
224 }
225
226 std::get<Major>(pt) += std::get<Major>(_step);
227 _error += _post_major_move;
228 }
229 };

243 template <class... Args>
244 static inline std::unique_ptr<scanner> make_scanner(point_t delta,
245 Args &&...args) {
246 const auto dx = std::get<0>(delta);
247 const auto dy = std::get<1>(delta);
248
249 scanner *impl;
250
251 if (std::abs(dx) > std::abs(dy)) { // major == 0
252 if (dy == 0)
253 impl = new axial_scanner<0>{delta,
254 std::forward<Args>(args)...};
255 else
256 impl = new bresenham_scanner<0>{
257 delta, std::forward<Args>(args)...};
258 }
259 else { // major == 1
260 if (dx == 0)
261 impl = new axial_scanner<1>{delta,
262 std::forward<Args>(args)...};
263 else
264 impl = new bresenham_scanner<1>{
265 delta, std::forward<Args>(args)...};
266 }
267
268 return std::unique_ptr<scanner>{impl};
269 }
