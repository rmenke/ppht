96 template <std::size_t Major>
97 class axial_scanner : public scanner {
98 static constexpr std::size_t Minor = 1 - Major;
99
100 const Point step;
101 const std::size_t radius;
102
103 public:
104 axial_scanner(Point delta, std::size_t radius)
105 : step{signum(std::get<0>(delta)), signum(std::get<1>(delta))}
106 , radius(radius) {
107 assert(std::get<Major>(step) != 0);
108 assert(std::get<Minor>(step) == 0);
109 }
110
111 void fill(Point point, std::set<Point> &points) const override {
112 std::get<Minor>(point) -= radius;
113
114 for (std::size_t m = 1; m < 2 * radius; ++m) {
115 ++std::get<Minor>(point);
116 points.insert(point);
117 }
118 }
119
120 void advance(Point &point) override {
121 std::get<Major>(point) += std::get<Major>(step);
122 }
123 };

142 template <std::size_t Major>
143 class bresenham_scanner : public scanner {
144 static constexpr std::size_t Minor = 1U - Major;
145
146 const Point _delta, _step, _perp_step;
147
148 const double _width;
149
150 const int_t _threshold =
151 std::get<Major>(_delta) - 2 * std::get<Minor>(_delta);
152 const int_t _post_minor_move = -2 * std::get<Major>(_delta);
153 const int_t _post_major_move = 2 * std::get<Minor>(_delta);
154
155 int_t _error = 0, _phase = 0;
156
157 void perpendiculars(Point point, std::set<Point> &points,
158 int_t initial_phase,
159 int_t initial_error) const {
160 auto p = point;
161 auto phase = initial_phase;
162
163 const auto d =
164 std::get<Major>(_delta) + std::get<Minor>(_delta);
165
166 for (auto tk = d - initial_error; tk < _width;
167 tk -= _post_minor_move) {
168 points.insert(p);
169
170 if (phase >= _threshold) {
171 std::get<Major>(p) += std::get<Major>(_perp_step);
172 phase += _post_minor_move;
173 tk += _post_major_move;
174 }
175
176 std::get<Minor>(p) += std::get<Minor>(_perp_step);
177 phase += _post_major_move;
178 }
179
180 p = point;
181 phase = -initial_phase;
182
183 for (auto tk = d + initial_error; tk <= _width;
184 tk -= _post_minor_move) {
185 points.insert(p);
186
187 if (phase > _threshold) {
188 std::get<Major>(p) -= std::get<Major>(_perp_step);
189 phase += _post_minor_move;
190 tk += _post_major_move;
191 }
192
193 std::get<Minor>(p) -= std::get<Minor>(_perp_step);
194 phase += _post_major_move;
195 }
196 }
197
198 public:
199 bresenham_scanner(Point delta, std::size_t radius)
200 : _delta{std::abs(std::get<0>(delta)),
201 std::abs(std::get<1>(delta))}
202 , _step{signum(std::get<0>(delta)), signum(std::get<1>(delta))}
203 , _perp_step{Major == 0 ? -std::get<0>(_step)
204 : +std::get<0>(_step),
205 Major == 0 ? +std::get<1>(_step)
206 : -std::get<1>(_step)}
207 , _width{2.0 * radius *
208 std::hypot(std::get<0>(_delta), std::get<1>(_delta))} {
209 }
210
211 void fill(Point point, std::set<Point> &points) const override {
212 perpendiculars(point, points, _phase, _error);
213
214 if (_error >= _threshold && _phase >= _threshold) {
215 std::get<Minor>(point) += std::get<Minor>(_step);
216 perpendiculars(
217 point, points,
218 (_phase + _post_minor_move + _post_major_move),
219 _error + _post_minor_move);
220 }
221
222 if (points.empty()) points.insert(point);
223 }
224
225 void advance(Point &pt) override {
226 if (_error >= _threshold) {
227 std::get<Minor>(pt) += std::get<Minor>(_step);
228 _error += _post_minor_move;
229
230 if (_phase >= _threshold) { _phase += _post_minor_move; }
231
232 _phase += _post_major_move;
233 }
234
235 std::get<Major>(pt) += std::get<Major>(_step);
236 _error += _post_major_move;
237 }
238 };

252 template <class... Args>
253 static inline std::unique_ptr<scanner> make_scanner(Point delta,
254 Args &&...args) {
255 const auto dx = std::get<0>(delta);
256 const auto dy = std::get<1>(delta);
257
258 scanner *impl;
259
260 if (std::abs(dx) > std::abs(dy)) { // major == 0
261 if (dy == 0)
262 impl = new axial_scanner<0>{delta,
263 std::forward<Args>(args)...};
264 else
265 impl = new bresenham_scanner<0>{
266 delta, std::forward<Args>(args)...};
267 }
268 else { // major == 1
269 if (dx == 0)
270 impl = new axial_scanner<1>{delta,
271 std::forward<Args>(args)...};
272 else
273 impl = new bresenham_scanner<1>{
274 delta, std::forward<Args>(args)...};
275 }
276
277 return std::unique_ptr<scanner>{impl};
278 }
